# 数据库设计

> **本文件（v1 schema）已被 `13-数据模型-v2` 取代。** 冲突处以 `11`~`16` 为准。
> 保留作历史输入，记录 v1 → v2 的演进。**本文件的 schema 不得直接使用**，原因：
>
> | # | v1 问题 | v2 处置 |
> |---|---|---|
> | 1 | generator 用 `prisma-client-js`；`datasource` 内写 `url` | Prisma 7 要求：generator 改 `prisma-client` 且 `output` 必填；`url` 迁至 `prisma.config.ts` |
> | 2 | 缺 9 个模型：`Page`、`Attachment`、`Role`、`Permission`、`RolePermission`、`UserRole`、`Account`、`Session`、`VerificationToken` | 全部补齐（缺口 1/2/6） |
> | 3 | `Article.status` 4 态；`AuditRecord` 无状态字段 | 6 态 + `fromStatus`/`toStatus`（A12） |
> | 4 | `User.siteId` 是裸字符串，`Site` 无反向关系 | 改真外键 + `Site.users`（缺口 5） |
> | 5 | `Article.slug` 可空且无唯一约束 | 必填 + `@@unique([siteId, slug])`（缺口 9） |
> | 6 | 无 `deletedAt`，回收站无法实现 | 5 个模型加软删除（缺口 8） |
> | 7 | 缺外键、缺索引；`Statistic` 的 `@@unique([dateKey, siteId])` 因 `siteId` 可空而失效 | 补外键 + 索引；`siteId` 改必填（缺口 10） |
> | 8 | `Comment` 无 `parentId`；`Media` 无 `siteId`/`folder`/`album`；`Navigation` 有 `parentId` 但无关系定义 | 全部补齐（缺口 3/4/14） |
> | 9 | `Article` 无"稿件归属人"字段 | 新增 `createdById`（状态机约束 C4 要求） |
>
> **权威来源**：[`13-数据模型-v2.md`](../13-数据模型-v2.md)（24 个 model，含完整 schema 与索引清单）

## Prisma Schema

```prisma
generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "sqlite"  // 生产切 postgresql
  url      = env("DATABASE_URL")
}

model Site {
  id        String    @id @default(cuid())
  name      String
  domain    String    @unique
  template  String    @default("default")
  logo      String?
  status    Boolean   @default(true)
  createdAt DateTime  @default(now())
  updatedAt DateTime  @updatedAt
  channels  Channel[]
  articles  Article[]
  navigations Navigation[]
}

model Channel {
  id        String    @id @default(cuid())
  siteId    String
  parentId  String?
  name      String
  slug      String
  type      String    @default("list") // list, page, link, form
  template  String?
  sort      Int       @default(0)
  status    Boolean   @default(true)
  createdAt DateTime  @default(now())
  updatedAt DateTime  @updatedAt
  site      Site      @relation(fields: [siteId], references: [id], onDelete: Cascade)
  parent    Channel?  @relation("ChannelTree", fields: [parentId], references: [id])
  children  Channel[] @relation("ChannelTree")
  articles  Article[]
}

model Article {
  id          String    @id @default(cuid())
  siteId      String
  channelId   String
  title       String
  slug        String?
  summary     String?
  content     String
  cover       String?
  author      String?
  source      String?
  tags        String?   // JSON 字符串
  status      String    @default("draft") // draft, pending, published, rejected
  top         Boolean   @default(false)
  recommend   Boolean   @default(false)
  viewCount   Int       @default(0)
  publishTime DateTime?
  createdAt   DateTime  @default(now())
  updatedAt   DateTime  @updatedAt
  site        Site      @relation(fields: [siteId], references: [id], onDelete: Cascade)
  channel     Channel   @relation(fields: [channelId], references: [id], onDelete: Cascade)
  versions    ArticleVersion[]
  audits      AuditRecord[]
  comments    Comment[]
}

model ArticleVersion {
  id        String   @id @default(cuid())
  articleId String
  version   Int
  title     String
  content   String
  editor    String
  createdAt DateTime @default(now())
  article   Article  @relation(fields: [articleId], references: [id], onDelete: Cascade)
}

model AuditRecord {
  id        String   @id @default(cuid())
  articleId String
  step      String   // submit, approve, reject, publish, withdraw
  operator  String
  role      String
  comment   String?
  createdAt DateTime @default(now())
  article   Article  @relation(fields: [articleId], references: [id], onDelete: Cascade)
}

model Media {
  id        String   @id @default(cuid())
  type      String   // image, video, file
  path      String
  name      String
  size      Int
  mimeType  String?
  width     Int?
  height    Int?
  uploader  String
  createdAt DateTime @default(now())
}

model User {
  id        String   @id @default(cuid())
  username  String   @unique
  password  String
  name      String
  email     String?
  avatar    String?
  role      String   @default("editor") // super_admin, site_admin, editor, auditor
  siteId    String?
  status    Boolean  @default(true)
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt
}

model Navigation {
  id       String  @id @default(cuid())
  siteId   String
  name     String
  url      String
  parentId String?
  sort     Int     @default(0)
  status   Boolean @default(true)
  site     Site    @relation(fields: [siteId], references: [id], onDelete: Cascade)
}

model Form {
  id        String   @id @default(cuid())
  name      String
  fields    String   // JSON
  status    Boolean  @default(true)
  createdAt DateTime @default(now())
  data      FormData[]
}

model FormData {
  id        String   @id @default(cuid())
  formId    String
  data      String   // JSON
  createdAt DateTime @default(now())
  form      Form     @relation(fields: [formId], references: [id], onDelete: Cascade)
}

model Comment {
  id        String   @id @default(cuid())
  articleId String
  name      String
  email     String?
  content   String
  status    String   @default("pending") // pending, approved, rejected
  createdAt DateTime @default(now())
  article   Article  @relation(fields: [articleId], references: [id], onDelete: Cascade)
}

model Message {
  id        String   @id @default(cuid())
  type      String   // leader, suggestion, complaint
  name      String
  contact   String?
  content   String
  status    String   @default("pending") // pending, processing, replied
  reply     String?
  createdAt DateTime @default(now())
}

model Statistic {
  id        String   @id @default(cuid())
  date      DateTime
  siteId    String?
  pv        Int      @default(0)
  uv        Int      @default(0)
  ip        Int      @default(0)
  source    String?
}

model Log {
  id        String   @id @default(cuid())
  type      String   // login, operation, audit, error
  userId    String?
  username  String?
  ip        String?
  action    String
  detail    String?
  createdAt DateTime @default(now())
}

model Config {
  id    String @id @default(cuid())
  key   String @unique
  value String
  group String @default("general")
}
